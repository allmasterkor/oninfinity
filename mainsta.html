// 1. 토큰 정화 로직이 추가된 Gist 로드 함수 async function testAndLoadGist() { // trim()을 붙여 앞뒤 공백 및 줄바꿈 문자 강제 제거 const token = document.getElementById('githubToken').value.trim(); const gistId = document.getElementById('gistId').value.trim(); const ticker = document.getElementById('ticker').value.trim(); if(!token || !gistId) return alert("설정 값을 모두 입력해주세요."); localStorage.setItem('gist_token', token); localStorage.setItem('gist_id', gistId); localStorage.setItem('gist_ticker', ticker); try { const response = await fetch(`https://api.github.com/gists/${gistId}`, { headers: { 'Authorization': `token ${token}` } }); if(!response.ok) throw new Error("Gist를 불러오지 못했습니다. 토큰이나 ID를 확인하세요."); const gistData = await response.json(); if (gistData.files[fileName]) { appData = JSON.parse(gistData.files[fileName].content); if(!appData.account.current_cycle) appData.account.current_cycle = 1; if(!appData.account.total_buy_amount) appData.account.total_buy_amount = 0; updateAccountUI(); updateReportUI(); alert("Gist 데이터 동기화 완료!"); } else { alert("Gist에 지정된 파일명(" + fileName + ")이 없습니다. 초기화 생성을 먼저 진행해주세요."); } } catch(e) { alert("에러 발생: " + e.message); } } // 2. 토큰 정화 로직이 추가된 Gist 쓰기 함수 async function saveToGist(gistId, token, data) { // 헤더에 들어가기 전 다시 한번 공백 제거 검증 const cleanToken = token.trim(); const cleanGistId = gistId.trim(); const response = await fetch(`https://api.github.com/gists/${cleanGistId}`, { method: 'PATCH', headers: { 'Authorization': `token ${cleanToken}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ files: { [fileName]: { content: JSON.stringify(data, null, 2) } } }) }); if(!response.ok) throw new Error("Gist 업로드 실패"); }
